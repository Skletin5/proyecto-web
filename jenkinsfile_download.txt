pipeline {
    agent any
    
    stages {
        stage('Clonar repositorio') {
            steps {
                git branch: 'main',
                    url: 'https://github.com/Skletin5/proyecto-web.git'
            }
        }
        stage('Construir contenedores'){
            steps {
                sh 'docker compose down'
                sh 'docker compose up -d --build'
            }
        }
        stage('Importación de cambios clonados') {
            steps {
                withCredentials([usernamePassword(credentialsId: 'DB_Data', passwordVariable: 'DB_ROOT_PASSWD', usernameVariable: 'DB_NAME')]) {
                    sh 'docker cp wp-content/. download_wordpress_1:/var/www/html/wp-content/'
                    sh 'docker exec download_wordpress_1 chown -R www-data:www-data /var/www/html/wp-content'
                    sh 'docker cp wp-config.php download_wordpress_1:/var/www/html/'
                    sh 'docker exec download_wordpress_1 chown www-data:www-data /var/www/html/wp-config.php'
                    sh "sed -i '/GTID/d' db-backup.sql"
                    sh 'sleep 15'
                    sh 'cat db-backup.sql | docker exec -i download_db_1 mysql -u root -p$DB_ROOT_PASSWD $DB_NAME'
                }
            }
        }
    }
}
